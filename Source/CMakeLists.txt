cmake_minimum_required(VERSION 3.16)
project(KPartitionSolutions LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../Exe")
foreach(name approximate_solution polynomial_solution exact_solution)
  add_executable(${name} "${name}/${name}.cpp")
  if(MINGW)
    target_link_options(${name} PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
endforeach()
add_executable(benchmark_suite benchmark_suite.cpp)
if(MINGW)
  target_link_options(benchmark_suite PRIVATE -static -static-libgcc -static-libstdc++)
endif()
